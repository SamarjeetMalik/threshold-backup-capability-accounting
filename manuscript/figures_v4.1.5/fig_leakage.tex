% Figure 2: leakage subspace (finite-field linear algebra, no geometry implied). v4.1.5: 8 pt sans, natural size.
\begin{tikzpicture}[x=1cm,y=1cm]
\node[node] (S) at (1.2,1.1) {seed $S\in D\subseteq\mathbb F_p^k$};
\node[neut] (R) at (1.2,0) {mask $R\sim\mathrm{Unif}(\mathbb F_p^r)$};
\node[node] (A) at (3.6,1.1) {$A\in\mathbb F_p^{m\times k}$};
\node[neut] (B) at (3.6,0) {$B\in\mathbb F_p^{m\times r}$};
\node[node] (Y) at (5.7,0.55) {view\\$Y=AS+BR$};
\draw[flow] (S)--(A); \draw[flow,cslate] (R)--(B);
\draw[flow] (A.east)--(Y.west); \draw[flow,cslate] (B.east)--(Y.west);
\node[node] (lam) at (8.2,0.55) {left kernel of $B$:\\every $\lambda$ with $\lambda B=0$};
\node[node] (MS) at (11.0,0.55) {$\lambda Y=\lambda A\,S=u\cdot S$\\with $u=\lambda A\in U$};
\draw[flow] (Y)--(lam); \draw[flow] (lam)--(MS);
\node[neut,anchor=north west,align=left] (rev) at (0,-0.8) {\textbf{What the view reveals}\\
$Y\mid S{=}s$ is uniform on $As+\operatorname{im}B$.\\
The view reveals exactly $MS$,\\where the rows of $M$ span $U$.\\
$I(S;Y)=H(MS)\le(\mathrm{rk}[A\;B]-\mathrm{rk}\,B)\log_2 p$.};
\node[adv,anchor=north east,align=left] (ex) at (12.45,-0.8) {\textcolor{cburg}{\textbf{Example: leakage without reconstruction}}\\
$Y=S_0+S_1$ (no mask): $A=[1\;1]$, $B$ empty.\\
$U=\operatorname{span}\{(1,1)\}$, so $e_0,e_1\notin U$:\\
neither seed is reconstructible,\\
yet $\dim U=1$ and $I(S;Y)=\log_2 p$ bits.};
\end{tikzpicture}
