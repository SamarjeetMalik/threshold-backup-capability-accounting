\begin{tikzpicture}[x=1cm,y=1cm,st/.style={node,minimum width=17mm,minimum height=6mm},bad/.style={adv,minimum width=17mm,minimum height=6mm},

% (a) path
\node[layer,anchor=west] at (-0.15,2.95) {(a) Path $(e,g)$};
\node[st] (sealed) at (0.85,2.0) {sealed};

\node[st] (er) at (7.6,2.0) {erased};
\draw[flow] (sealed)--(exp) node[midway,lab,above=9pt]{BeginExpiry};

\draw[flow] (send)--(done) node[midway,lab,above=9pt]{Transfer};

\draw[aflow] ([xshift=-5.5mm]dec.south) -- (ab.north) node[midway,left=2pt,text=cburg,font=\figfont,align=right]{expiry or\\config.\ change};

\draw[flow] ([xshift=6mm]dec.south) |- ([yshift=2.2pt]can.west) node[pos=0.75,lab,above=1pt]{WorkerAbandons};
\draw[flow] ([yshift=-2.2pt]ab.east) -- ([yshift=-2.2pt]can.west) node[midway,lab,below=1pt]{WorkerReturns};
% queued and sending are cancelled by expiry, configuration change or crash (grey)
\draw[flow,cslate] (que.south) |- (6.35,-1.5) -| ([xshift=-3mm]can.north);
\draw[flow,cslate] (send.south) |- (9.1,-1.5) -| ([xshift=3mm]can.north);
\node[font=\figfont,text=cslate,align=center] at (7.75,-1.0) {expiry, config.\\change or crash};
